\chapter{QUÁ TRÌNH THIẾT KẾ VÀ XÂY DỰNG SẢN PHẨM}
\section{Cơ sở lý thuyết}

\subsection{Kiến trúc Transformer}

Transformer được áp dụng để xử lý chuỗi đặc trưng tay theo thời gian. Thành phần cốt lõi là Scaled Dot-Product Attention:

\begin{equation}
    \text{Attention}(\mathbf{Q}, \mathbf{K}, \mathbf{V}) 
    = \text{softmax}\!\left(\frac{\mathbf{Q}\mathbf{K}^\top}{\sqrt{d_k}}\right)\mathbf{V}
\end{equation}

Multi-Head Attention với $h = 8$ heads, $d_\text{model} = 256$, 4 lớp Pre-LayerNorm:

\begin{equation}
    \text{head}_i = \text{Attention}(\mathbf{Q}\mathbf{W}_i^Q,\; \mathbf{K}\mathbf{W}_i^K,\; \mathbf{V}\mathbf{W}_i^V), \qquad
    \text{MultiHead} = \text{Concat}(\text{head}_1, \ldots, \text{head}_h)\,\mathbf{W}^O
\end{equation}

Positional Encoding mã hóa vị trí frame ($T = 64$):

\begin{equation}
    \text{PE}_{(pos,\,2i)} = \sin\!\left(\frac{pos}{10000^{2i/d}}\right), \qquad
    \text{PE}_{(pos,\,2i+1)} = \cos\!\left(\frac{pos}{10000^{2i/d}}\right)
\end{equation}

\subsection{Graph Attention Network (GAT)}

GAT học quan hệ không gian giữa 6 nhóm ngón tay theo cấu trúc giải phẫu. Ma trận kề chuẩn hóa $\hat{\mathbf{A}} = \mathbf{D}^{-1}(\mathbf{A} + \mathbf{I})$, attention chỉ tính trên cạnh hợp lệ ($h = 4$ heads, finger\_dim $= 32$):

\begin{equation}
    \alpha_{ij} = \frac{\exp(s_{ij}) \cdot \hat{A}_{ij}}{\sum_{k} \exp(s_{ik}) \cdot \hat{A}_{ik}}, \qquad
    \mathbf{h}_i' = \text{LayerNorm}\!\left(\mathbf{W}^O\sum_j \alpha_{ij}\mathbf{V}_j + \mathbf{h}_i\right)
\end{equation}

\subsection{Cross-Attention và Temporal Pooling}

Cross-Attention hai chiều học quan hệ tay trái $\leftrightarrow$ tay phải:

\begin{equation}
    \mathbf{L}' = \text{LayerNorm}(\mathbf{L} + \text{Attention}(\mathbf{L},\mathbf{R},\mathbf{R})), \qquad
    \mathbf{R}' = \text{LayerNorm}(\mathbf{R} + \text{Attention}(\mathbf{R},\mathbf{L},\mathbf{L}))
\end{equation}

Sau Temporal Transformer, chuỗi được tổng hợp bằng learned attention pooling:

\begin{equation}
    w_t = \frac{\exp(\mathbf{h}_t \mathbf{w})}{\sum_{t'} \exp(\mathbf{h}_{t'} \mathbf{w})}, \qquad
    \mathbf{c} = \sum_{t=1}^{T} w_t \cdot \mathbf{h}_t
\end{equation}

\subsection{Hàm mất mát và tối ưu hóa}

Label Smoothing Cross-Entropy ($\varepsilon = 0.1$, $C = 108$):

\begin{equation}
    \mathcal{L} = -\sum_{c=1}^{C} \tilde{y}_c \log \hat{y}_c, \qquad
    \tilde{y}_c = \begin{cases}
        1 - \varepsilon + \varepsilon/C & c = y^* \\
        \varepsilon/C                   & c \neq y^*
    \end{cases}
\end{equation}

Tối ưu hóa bằng AdamW ($\lambda = 10^{-4}$), Linear Warmup 8 epoch rồi Cosine Decay về $\alpha_\text{min} = 10^{-6}$, Early Stopping patience $= 20$.

\subsection{Data Augmentation}

Tập dữ liệu $108 \times 100 = 10.800$ mẫu, augment 9 lần ($p = 0.4$) bằng Gaussian Noise lên tọa độ tay và đặc trưng curl. Tổng mẫu hiệu dụng: $97.200$.


\section{Quy trình chuyển đổi tiếng Việt sang ngôn ngữ ký hiệu (VN $\rightarrow$ VSL)}

Đây là chiều "thuận" của hệ thống, giúp người nghe/nói thông thường có thể
truyền đạt thông điệp tới người khiếm thính thông qua thiết bị.

\subsection{Quy trình cơ bản}

Thiết bị thu nhận đầu vào là giọng nói. Tín hiệu âm thanh được chuyển đổi thành
văn bản thông qua module STT, tạo thành văn bản thô đại diện
cho nội dung người nói muốn truyền đạt. Văn bản này sau đó được gửi lên server
để xử lý qua bộ Logic Solving. Phản hồi từ server bao gồm danh sách các từ đã
ánh xạ cùng URL video tương ứng, được gửi trả về thiết bị và hiển thị dưới dạng
giao diện kèm animation chuyển động 3D hoặc video người ký hiệu thực sự.

\subsection{Quy trình logic xử lý chi tiết}

Khi nhận được văn bản đầu vào (ví dụ: "Tôi xơi cơm với bạn hồi chiều hôm qua"),
hệ thống tiến hành qua năm bước xử lý tuần tự:

\begin{itemize}
    \item \textbf{Bước 1 -- Chuẩn hóa dữ liệu:} Toàn bộ văn bản được chuyển về
    chữ thường, loại bỏ dấu câu, ký tự đặc biệt và khoảng trắng thừa. Bước này
    đảm bảo tính nhất quán khi tra cứu từ điển.

    \item \textbf{Bước 2 -- Tách từ (Word Segmentation):} Thư viện Underthesea
    được sử dụng để phân tách câu tiếng Việt thành các từ/cụm từ có nghĩa. Đây
    là bước thiết yếu vì tiếng Việt không phân cách từ bằng khoảng trắng như
    tiếng Anh.

    \item \textbf{Bước 3 -- Quét cửa sổ trượt (Sliding Window):} Thay vì dịch
    từng từ đơn lẻ, hệ thống ưu tiên tìm kiếm các cụm từ dài nhất có trong từ
    điển ký hiệu trước, rồi dần thu hẹp xuống. Ví dụ, với chuỗi "xơi cơm", hệ
    thống ánh xạ trực tiếp sang ký hiệu "ăn cơm" thay vì tách thành hai ký hiệu
    rời rạc, đảm bảo tính tự nhiên và tránh dịch máy móc.

    \item \textbf{Bước 4 -- Lọc từ dừng (Stopwords Filtering):} Các từ nối không
    mang nghĩa thực thể trong ngôn ngữ ký hiệu như "với", "hồi", "của", "thì",
    "mà"... nếu không tìm thấy trong từ điển ký hiệu sẽ bị bỏ qua, giúp câu ký
    hiệu trở nên súc tích và đúng cú pháp VSL hơn.

    \item \textbf{Bước 5 -- Sắp xếp lại theo cú pháp VSL:} Các từ/cụm từ đã lọc
    được sắp xếp lại theo cấu trúc chuẩn của VSL (THỜI GIAN $\rightarrow$ ĐỊA
    ĐIỂM $\rightarrow$ CHỦ THỂ $\rightarrow$ ĐỘNG TÁC $\rightarrow$ ĐỐI TƯỢNG),
    sau đó được ánh xạ sang danh sách URL video tương ứng trong cơ sở dữ liệu và
    trả về thiết bị để phát.
\end{itemize}

\section{Quy trình chuyển đổi ngôn ngữ ký hiệu sang tiếng Việt (VSL $\rightarrow$ VN)}

Đây là chiều "nghịch" và phức tạp hơn, đòi hỏi hệ thống phải hiểu và giải mã
các chuỗi cử chỉ động trong không gian ba chiều từ Quy trình video camera.

\subsection{Quy trình cơ bản}

Thiết bị tiếp nhận Quy trình frames video liên tục từ camera và gửi lên server.
Server thực hiện toàn bộ quá trình Logic Solving bao gồm trích xuất đặc trưng,
nhận diện cử chỉ, lọc cảm xúc và xử lý ngữ cảnh, sau đó trả về kết quả phiên
dịch dưới dạng văn bản và giọng đọc để hiển thị trên giao diện người dùng.

\subsection{Thu thập dữ liệu huấn luyện}

Dữ liệu huấn luyện được thu thập từ webcam theo quy trình có kiểm soát và có
cấu trúc. Người thu thập lần lượt chọn nhãn ký hiệu cần quay, chọn trạng thái
cảm xúc kết hợp trong số 7 trạng thái được định nghĩa, sau đó thực hiện ký hiệu
trong khoảng thời gian tối đa 10 giây trước khi lưu lại. Mỗi phiên quay tạo ra
một cặp file gồm video và file metadata đi kèm.

File metadata lưu trữ toàn bộ thông tin ngữ cảnh của video bao gồm trạng thái
cảm xúc, mã định danh cảm xúc và ngày cập nhật. Nhờ đó, dữ liệu có thể được
truy vết, kiểm tra chất lượng và tái sử dụng một cách dễ dàng. Trong suốt quá
trình quay, giao diện hiển thị các điểm mốc bàn tay theo thời gian thực, giúp
người thu thập quan sát và điều chỉnh tư thế tay ngay lập tức, đảm bảo chất
lượng dữ liệu đầu vào từ đầu.

\subsection{Trích xuất đặc trưng}

Các file video thô được chuyển đổi sang định dạng mảng số để phục vụ huấn luyện
mô hình. Quy trình xử lý bao gồm các bước: đọc từng frame của video $\rightarrow$
đưa qua MediaPipe để phát hiện điểm mốc $\rightarrow$ chuẩn hóa tọa độ
$\rightarrow$ resample về chuỗi cố định 64 frames $\rightarrow$ ghép vector cảm
xúc one-hot $\rightarrow$ lưu file với kích thước \textbf{(64, 208)}.

Cấu trúc feature vector \textbf{208 chiều} cho mỗi frame bao gồm:

\begin{itemize}
    \item \textbf{Pose (45 chiều):} 15 điểm xương quan trọng trên cơ thể (mũi,
    vai, khuỷu tay, cổ tay, ngón tay cái, hông...), mỗi điểm gồm 3 tọa độ
    (x, y, z). Dữ liệu được chuẩn hóa bằng cách lấy điểm giữa hông làm tâm và
    chia cho khoảng cách giữa hai vai, đảm bảo bất biến về vị trí và kích thước
    người dùng.

    \item \textbf{Hands (126 chiều):} 21 điểm landmark cho mỗi bàn tay trong
    không gian 3 chiều ($21 \times 3 \times 2$ tay = 126 chiều). Khi chỉ phát
    hiện được một tay, hệ thống luôn gán vào slot 0 bất kể trái hay phải. Khi
    có hai tay, các slot được sắp xếp theo tọa độ x-center (tay bên trái màn
    hình vào slot 0, bên phải vào slot 1). Tọa độ được chuẩn hóa với gốc tại
    cổ tay và chia cho khoảng cách MCP-9.

    \item \textbf{Finger Curl (30 chiều):} Mô tả mức độ co duỗi của từng ngón
    tay trên cả hai bàn tay ($5$ ngón $\times$ 3 đặc trưng $\times$ 2 tay = 30
    chiều). Mỗi ngón được đặc trưng bởi ba giá trị: \textit{curl\_ratio} (tỉ lệ
    khoảng cách đầu ngón so với cổ tay, 0 = co hoàn toàn, 1 = duỗi thẳng),
    \textit{bend\_angle} (góc gập tại khớp PIP chuẩn hóa, 0 = thẳng, 1 = co tối
    đa), và \textit{tip\_dist\_norm} (khoảng cách đầu ngón đến cổ tay, chuẩn hóa
    theo kích thước bàn tay). Thành phần này cho phép phân biệt các ký hiệu có
    hình dạng bàn tay tổng thể giống nhau nhưng khác nhau ở trạng thái uốn cong
    của từng ngón.

    \item \textbf{Emotion (7 chiều):} Vector one-hot encoding biểu diễn 7 trạng
    thái cảm xúc: Angry (giận dữ), Disgust (ghê tởm), Fear (sợ hãi), Happy (vui
    vẻ), Sad (buồn bã), Surprise (ngạc nhiên) và Neutral (trung tính). Trạng
    thái cảm xúc được xác định từ ảnh khuôn mặt thông qua mô hình FER được huấn
    luyện trên bộ dữ liệu FER-2013.
\end{itemize}

Để tăng cường dữ liệu, mỗi video gốc được biến thể thành \textbf{9 file}: 1 file
gốc, 1 file mirror hoán đổi tay trái $\leftrightarrow$ tay phải, 1 file mirror
kết hợp nhiễu Gaussian, 2 file thêm nhiễu Gaussian vào tọa độ tay (mức 0.003 và
0.006), 2 file co giãn tỷ lệ tay ($\times$0.95 và $\times$1.05), và 2 file time
warp (cắt 5\% và 10\% đầu/cuối). Chiến lược này \textbf{chỉ áp dụng cho tập
Train}, không áp dụng cho Val và Test, nhằm đánh giá khả năng tổng quát hóa thực
sự của mô hình.

\subsection{Huấn luyện mô hình Hand-Aware Transformer}

Mô hình được sử dụng là \textbf{HandAwareVSLClassifier} -- một kiến trúc
Transformer có nhận thức về cấu trúc bàn tay, được thiết kế để khai thác đồng
thời quan hệ không gian giữa các ngón tay và quan hệ thời gian giữa các frame.
Đầu vào có kích thước \textbf{(B, 64, 208)}, được xử lý qua 6 module tuần tự:

\begin{itemize}
    \item \textbf{Feature Parser:} Tách feature vector 208 chiều thành các thành
    phần riêng biệt -- pose (45), tọa độ tay trái (63), tọa độ tay phải (63),
    curl tay trái (15), curl tay phải (15) và emotion (7) -- để xử lý chuyên
    biệt theo từng loại đặc trưng.

    \item \textbf{Finger Encoder:} Encode từng nhóm ngón tay độc lập (cổ tay và
    5 ngón = 6 nhóm) bằng MLP riêng cho mỗi nhóm, tạo ra biểu diễn không gian
    với \texttt{finger\_dim = 32}. Cùng một bộ encoder được dùng chung cho cả
    hai tay.

    \item \textbf{Finger Graph Attention:} Áp dụng Graph Attention Network
    với ma trận kề phản ánh cấu trúc giải phẫu thực của bàn tay -- cổ tay kết
    nối với 5 ngón, các ngón kề nhau được kết nối. Cơ chế attention chỉ cho phép
    thông tin truyền qua các cạnh giải phẫu hợp lệ, giúp mô hình học được quan
    hệ không gian có ý nghĩa sinh học. Chạy độc lập cho tay trái và tay phải với
    4 attention heads.

    \item \textbf{Hand Cross Attention:} Cross-attention hai chiều học quan hệ
    tương tác giữa tay trái và tay phải. Tay trái attend vào tay phải và ngược
    lại, cho phép mô hình nắm bắt các ký hiệu hai tay phối hợp.

    \item \textbf{Curl Encoder và Hand Aggregator:} Curl Encoder encode đặc trưng
    co duỗi của từng ngón thành embedding 32 chiều cho mỗi tay. Hand Aggregator
    sau đó gộp toàn bộ các thành phần -- finger embedding, curl embedding, pose
    và emotion -- thành vector \textbf{256 chiều} cho mỗi timestep, sẵn sàng
    đưa vào bước xử lý thời gian.

    \item \textbf{Temporal Transformer:} Transformer encoder với chuẩn hóa
    trước (pre-norm), 4 lớp và 8 attention heads, xử lý chuỗi 64 timestep để
    học các phụ thuộc thời gian dài hạn. Positional encoding dạng sinusoidal
    được thêm vào trước khi đưa vào encoder. Sau Transformer, cơ chế
    \textbf{Temporal Attention Pooling} học trọng số tầm quan trọng của từng
    frame và tổng hợp thành vector ngữ cảnh toàn cục, cuối cùng được đưa qua
    bộ phân loại MLP 3 lớp để dự đoán nhãn ký hiệu.
\end{itemize}

Các siêu tham số chính của mô hình: finger\_dim = 32, curl\_embed\_dim = 32,
temporal\_dim = 256, temporal\_layers = 4, temporal\_heads = 8, graph\_heads = 4,
dropout = 0.2, label smoothing = 0.1, optimizer AdamW với learning rate
$5 \times 10^{-4}$, warmup tuyến tính 8 epoch, cosine decay và gradient
clipping = 1.0.

\subsection{Nhận dạng theo thời gian thực}

Hệ thống nhận dạng theo thời gian thực hoạt động theo vòng lặp liên tục gồm 10
bước: (1) thu frame BGR từ webcam, (2) lật ảnh và chuyển sang RGB, (3) phát hiện
điểm mốc bất đồng bộ qua MediaPipe, (4) trích xuất \textbf{201 chiều} đặc trưng
pose, tọa độ tay và finger curl, (5) chuẩn hóa frame, (6) ghép vector cảm xúc
tạo thành vector \textbf{208 chiều}, (7) đưa vào buffer cửa sổ trượt, (8) khi
đủ 64 frame liên tiếp thì thực hiện suy luận, (9) áp dụng softmax lấy nhãn và
điểm tin cậy, (10) kiểm tra tính ổn định trong 3 frame.

Ba cơ chế tối ưu quan trọng được áp dụng:

\begin{itemize}
    \item \textbf{Xử lý bất đồng bộ (MediaPipe LIVE\_STREAM mode):} Phát hiện
    điểm mốc chạy song song với Quy trình thu video thông qua cơ chế callback
    thread-safe, tránh hiện tượng mất frame kể cả khi tài nguyên phần cứng hạn
    chế.

    \item \textbf{Bộ đệm cửa sổ trượt (Sliding Window Buffer):} Bộ đệm tự động
    loại bỏ frame cũ nhất khi thêm frame mới, duy trì cửa sổ quan sát 64 frame
    gần nhất và cho phép dự đoán liên tục mà không cần chờ kết thúc đoạn cử chỉ
    hoàn chỉnh.

    \item \textbf{Bộ lọc ổn định (Stability Filter):} Kết quả chỉ được hiển thị
    khi điểm tin cậy $\geq$ 0.6 và cùng nhãn xuất hiện trong ít nhất 3 frame
    liên tiếp. Cơ chế này triệt tiêu hiện tượng nhãn bị giật/nhảy liên tục do
    nhiễu, đảm bảo trải nghiệm người dùng mượt mà.
\end{itemize}

\section{Vai trò then chốt của Emotion Encoding trong NNKH}

Một trong những đóng góp quan trọng và đặc sắc nhất của đề tài là việc tích hợp Emotion Encoding vào feature vector. Trong Ngôn ngữ Ký hiệu, biểu cảm khuôn mặt không chỉ là yếu tố thứ cấp mà là một thành phần ngữ pháp bắt buộc, có khả năng thay đổi hoàn toàn ý nghĩa của một cử chỉ.


Ví dụ minh họa: Cùng một cử chỉ tay nhưng khi kết hợp với nét mặt NEUTRAL sẽ có nghĩa là "Cảm ơn" (thông thường), trong khi nét mặt HAPPY biểu đạt "Cảm ơn chân thành" (vui vẻ), nét mặt ANGRY lại dẫn đến ý nghĩa tiêu cực như "Ghét" hoặc sự bực bội, còn SURPRISE truyền đạt sự ngạc nhiên hoàn toàn khác.

Việc thêm 7 chiều emotion vào mỗi timestep (từ 171 lên 178 chiều) không chỉ tăng dung lượng thông tin mà còn cung cấp cho mô hình một lớp ngữ cảnh quan trọng mà các hệ thống chỉ dựa trên nhận diện tay thuần túy hoàn toàn bỏ sót. Đây là điểm phân biệt và tạo ra ưu thế cạnh tranh rõ rệt của hệ thống này so với các công trình trước đó.

\subsection{Cơ chế trích xuất và mã hóa cảm xúc khuôn mặt}

Hệ thống phân tích biểu cảm khuôn mặt dựa trên hai nguồn đầu ra song song từ
MediaPipe FaceLandmarker chạy ở chế độ \texttt{LIVE\_STREAM}: 468 landmarks tọa
độ 3D biểu diễn hình học khuôn mặt, và 52 blendshape coefficients biểu diễn mức
độ kích hoạt của từng nhóm cơ mặt theo thang $[0, 1]$.

Module \texttt{FacialExpressionAnalyzer} ưu tiên phân tích theo blendshapes thông
qua hàm \texttt{analyze\_blendshapes()}, tổng hợp các chỉ số chức năng gồm độ mở
miệng, nụ cười, độ mở hai mắt, mắt mở to, mắt nheo, và trạng thái lông mày. Các
chỉ số này được ánh xạ theo quy tắc ngưỡng sang một trong 7 nhãn cảm xúc chuẩn.
Khi blendshapes không khả dụng do điều kiện ánh sáng hoặc góc quay, hệ thống tự
động chuyển sang \texttt{analyze\_landmarks()} --- thực hiện phân tích hình học
thuần túy từ khoảng cách tương đối giữa các landmarks để ước tính cùng tập chỉ
số trên. Cơ chế dự phòng này đảm bảo nhãn cảm xúc được ghi nhận liên tục trong
suốt quá trình thu thập.

Tập nhãn cảm xúc gồm 7 lớp được định nghĩa thống nhất qua hằng số
\texttt{EMOTIONS}: \textit{angry} (0), \textit{disgust} (1), \textit{fear} (2),
\textit{happy} (3), \textit{sad} (4), \textit{surprise} (5), \textit{neutral}
(6). Tập nhãn này được dùng xuyên suốt toàn bộ pipeline --- từ giao diện chọn
nhãn khi thu video (\texttt{ask\_emotion()}), lưu metadata
(\texttt{save\_video\_emotion()}), cho đến nạp dữ liệu huấn luyện trong
\texttt{video\_to\_npy.py}.

\subsection{Tích hợp vào feature vector}

Sau khi xác định nhãn cảm xúc, hệ thống thực hiện mã hóa one-hot thành vector
7 chiều và nối vào cuối feature vector chính tại mỗi timestep. Feature vector
chính bao gồm 201 chiều từ pose landmarks, hand landmarks và finger curl. Kết
quả là tensor đầu vào cuối cùng có kích thước \textbf{208 chiều} $\times$ $T$
timestep, trong đó 7 chiều emotion cung cấp ngữ cảnh biểu cảm bổ sung cho từng
khung hình trong chuỗi ký hiệu.

Thông tin emotion được lưu song song với video dưới dạng file \texttt{.json} kèm
theo, ghi nhận cả tên nhãn lẫn chỉ số nguyên (\texttt{emotion\_id}) để đảm bảo
tính tương thích giữa các bước trong pipeline xử lý dữ liệu.

Cách tiếp cận này khác biệt so với các hệ thống CNN nhận diện cảm xúc học từ
pixel ảnh khuôn mặt: thay vì xây dựng một nhánh CNN riêng biệt đòi hỏi lượng
lớn dữ liệu gán nhãn, hệ thống tận dụng biểu diễn trung gian chất lượng cao của
MediaPipe vốn đã được tối ưu hóa trên tập dữ liệu quy mô lớn, nhờ đó giảm đáng
kể chi phí huấn luyện trong khi vẫn đảm bảo độ chính xác cần thiết cho bài toán
nhận dạng ngôn ngữ ký hiệu.

\section{Cơ chế lựa chọn ứng viên tiềm năng -- Pipeline 4 giai đoạn}

Để nâng cao độ chính xác cuối cùng và tính tự nhiên của kết quả phiên dịch, hệ
thống áp dụng một pipeline 4 giai đoạn lọc dần từ nhiều ứng viên xuống một kết
quả chính xác nhất:

\begin{itemize}
    \item \textbf{Giai đoạn 1 -- Mô hình cử chỉ $\rightarrow$ Top-3:} Mô hình
    HandAwareVSLClassifier đưa ra 3 từ ứng viên có xác suất cao nhất dựa trên
    đặc trưng cử chỉ tay và cấu trúc ngón.

    \item \textbf{Giai đoạn 2 -- Lọc cảm xúc $\rightarrow$ Top-2:} Lọc tiếp 3
    ứng viên bằng kết quả nhận diện biểu cảm khuôn mặt, loại bỏ các từ không
    phù hợp với cảm xúc đang biểu đạt, còn lại 2 ứng viên.

    \item \textbf{Giai đoạn 3 -- Làm mượt thời gian $\rightarrow$ Top-2:} Áp
    dụng làm mượt theo thời gian qua nhiều frame để loại bỏ dự đoán bất thường,
    giữ lại 2 ứng viên ổn định nhất.

    \item \textbf{Giai đoạn 4 -- Chấm điểm ngữ cảnh bằng PhoBERT $\rightarrow$
    Kết quả cuối:} Mô hình ngôn ngữ PhoBERT đánh giá xác suất xuất hiện của
    từng ứng viên trong ngữ cảnh các từ đã được nhận diện trước đó trong câu,
    chọn ra từ phù hợp nhất cả về cử chỉ, cảm xúc lẫn ngữ cảnh tiếng Việt.
\end{itemize}
